import datetime, hashlib, json, pathlib, subprocess, tarfile, tempfile
import yaml
root = pathlib.Path('/home/kaalin/.cache/wamn-lanes/wasmcloud-2-9-20260909')
upstream = pathlib.Path('/home/kaalin/.cache/wamn-lanes/upstream-wasmcloud-2-9-20260909')
out = root / 'docs/perf/2026.09/wasmcloud-2-9-cutover/deployment-crds-001'
out.mkdir(parents=True, exist_ok=True)
chart = 'oci://ghcr.io/wasmcloud/charts/runtime-operator'
expected = 'sha256:d70b240cfc3c745f306fc6ebecebff4370e6c8c0568b55c1d02b1eda1716fd17'
commands = []
def write(name, value):
    (out / name).write_text(json.dumps(value, indent=2) + '\n')
def run(name, argv):
    started = datetime.datetime.now(datetime.timezone.utc).isoformat()
    result = subprocess.run(argv, cwd=root, capture_output=True, timeout=55)
    commands.append({'name': name, 'argv': argv, 'cwd': str(root), 'started': started, 'finished': datetime.datetime.now(datetime.timezone.utc).isoformat(), 'exit': result.returncode})
    write('commands.json', commands)
    (out / (name + '.stdout')).write_bytes(result.stdout)
    (out / (name + '.stderr')).write_bytes(result.stderr)
    result.check_returncode()
    return result

def sha(raw):
    return hashlib.sha256(raw).hexdigest()

try:
    inventory = {}
    documents = {}
    identities = {}
    with tempfile.TemporaryDirectory(prefix='wamn-distributed-crds-') as temp:
        for version in ['2.9.0', '2.8.0']:
            destination = pathlib.Path(temp) / version
            destination.mkdir()
            result = run('helm-pull-' + version, ['helm', 'pull', chart, '--version', version, '--destination', str(destination)])
            lines = (result.stdout + b'\n' + result.stderr).decode().splitlines()
            digests = [line.removeprefix('Digest: ').strip() for line in lines if line.startswith('Digest: ')]
            assert len(digests) == 1, lines
            if version == '2.9.0':
                assert digests[0] == expected, digests
            package = destination / ('runtime-operator-' + version + '.tgz')
            crds = {}
            documents[version] = {}
            with tarfile.open(package, 'r:gz') as archive:
                metadata = yaml.safe_load(archive.extractfile('runtime-operator/Chart.yaml').read())
                assert metadata['version'] == metadata['appVersion'] == version
                for entry in archive.getmembers():
                    if not entry.isfile() or not entry.name.startswith('runtime-operator/crds/'):
                        continue
                    raw = archive.extractfile(entry).read()
                    document = yaml.safe_load(raw)
                    assert document['kind'] == 'CustomResourceDefinition'
                    assert document['apiVersion'] == 'apiextensions.k8s.io/v1'
                    documents[version][document['metadata']['name']] = document
                    spec = document['spec']
                    canonical_spec = json.dumps(spec, sort_keys=True, separators=(',', ':'), ensure_ascii=False).encode()
                    crds[document['metadata']['name']] = {
                        'chart_path': entry.name,
                        'api_version': document['apiVersion'],
                        'kind': document['kind'],
                        'name': document['metadata']['name'],
                        'resource_kind': spec['names']['kind'],
                        'scope': spec['scope'],
                        'versions': [{'name': v['name'], 'served': v['served'], 'storage': v['storage']} for v in spec['versions']],
                        'file_sha256': sha(raw),
                        'spec_sha256': sha(canonical_spec),
                    }
                    if version == '2.9.0':
                        source = upstream / 'charts' / entry.name
                        crds[document['metadata']['name']]['source_file_sha256'] = sha(source.read_bytes())
                        crds[document['metadata']['name']]['source_file_byte_identical'] = source.read_bytes() == raw
                        assert crds[document['metadata']['name']]['source_file_byte_identical']
            assert len(crds) == 5
            shown = run('helm-show-crds-' + version, ['helm', 'show', 'crds', str(package)])
            shown_documents = list(yaml.safe_load_all(shown.stdout))
            assert len(shown_documents) == 5
            assert all(isinstance(d, dict) and d.get('kind') == 'CustomResourceDefinition' for d in shown_documents)
            assert {d['metadata']['name']: d for d in shown_documents} == documents[version]
            inventory[version] = crds
            identities[version] = {'chart': chart, 'version': version, 'app_version': metadata['appVersion'], 'oci_manifest_digest': digests[0], 'package_sha256': sha(package.read_bytes())}
    assert inventory['2.9.0'].keys() == inventory['2.8.0'].keys()
    def changes(old, new, path=''):
        if isinstance(old, dict) and isinstance(new, dict):
            result = []
            for key in sorted(old.keys() | new.keys()):
                if key not in old:
                    result.append({'change': 'added', 'path': path + '/' + key})
                elif key not in new:
                    result.append({'change': 'removed', 'path': path + '/' + key})
                else:
                    result.extend(changes(old[key], new[key], path + '/' + key))
            return result
        if isinstance(old, list) and isinstance(new, list) and len(old) == len(new):
            return [item for i, (before, after) in enumerate(zip(old, new)) for item in changes(before, after, path + '/' + str(i))]
        return [] if old == new else [{'change': 'changed', 'path': path}]
    schema_changes = {name: changes(documents['2.8.0'][name]['spec'], documents['2.9.0'][name]['spec']) for name in sorted(documents['2.9.0'])}
    assert sum(len(items) for items in schema_changes.values()) == 6
    assert all(item['change'] == 'added' and item['path'].rsplit('/', 1)[1] in ['reclaimMinInstances', 'reclaimWindowSeconds'] for items in schema_changes.values() for item in items)
    write('schema-changes.json', schema_changes)
    comparison = [{'name': name, 'file_byte_identical': inventory['2.9.0'][name]['file_sha256'] == inventory['2.8.0'][name]['file_sha256'], 'spec_identical': inventory['2.9.0'][name]['spec_sha256'] == inventory['2.8.0'][name]['spec_sha256']} for name in sorted(inventory['2.9.0'])]
    write('distributed-chart-identities.json', identities)
    write('crd-inventory.json', {'spec_hash_encoding': 'UTF-8 JSON of CRD .spec, sorted keys, compact separators, ensure_ascii=False', 'source_commit': '68ebece9c537f8bb4b5c9999f274ec68d60f35a9', 'versions': inventory})
    write('comparison.json', {'distributed_2_9_vs_2_8': comparison, 'all_five_files_byte_identical': all(row['file_byte_identical'] for row in comparison), 'all_five_specs_identical': all(row['spec_identical'] for row in comparison), 'source_2_9_vs_distributed_2_9': 'all five CRD files byte-identical', 'deployment_001_operator_render': 'selected operator templates only; no CRDs', 'cluster_apply': 'not executed', 'installed_crd_schema_identity': 'not inspected or proven', 'gateway': 'unchanged, disabled in WAMN values'})
    write('upstream-recipe.json', {'source_commit': '68ebece9c537f8bb4b5c9999f274ec68d60f35a9', 'readme': {'path': 'runtime-operator/README.md', 'line': 27, 'instruction': 'make install'}, 'makefile': {'path': 'runtime-operator/Makefile', 'line': 155, 'command': '$(KUSTOMIZE) build config/crd | $(KUBECTL) apply --server-side=true --force-conflicts -f -'}, 'wamn_procedure': 'Apply CRDs from the downloaded chart with server-side apply and field manager wamn-cutover; no --force-conflicts; stop on a nonzero apply exit.', 'helm_crd_behavior_source': 'https://helm.sh/docs/chart_best_practices/custom_resource_definitions/', 'helm_crd_behavior': 'Helm installs missing crds/ declarations, skips existing declarations, and does not upgrade CRDs.'})
    write('capture-result.json', {'exit': 0, 'cluster_access': False})
    print(json.dumps({'exit': 0, 'comparison': comparison}, indent=2))
except Exception as error:
    write('capture-result.json', {'exit': 1, 'error': str(error), 'cluster_access': False})
    raise
finally:
    (out / 'capture-command.py.txt').write_text(pathlib.Path(__file__).read_text())
