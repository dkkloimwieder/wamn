def redact(value):
    value = re.sub(r'(https?://)[^/\s:@]+:[^@/\s]+@', r'\1[REDACTED]@', value)
    return re.sub(r'(?i)(Bearer\s+)[A-Za-z0-9._~+/-]+=*', r'\1[REDACTED]', value)


def run(args, cwd=ROOT, retain=True):
    try:
        result = subprocess.run(args, cwd=cwd, capture_output=True, timeout=20)
        entry = {'argv': args, 'cwd': str(cwd), 'exit_code': result.returncode}
        if retain:
            entry.update(stdout=redact(result.stdout.decode()), stderr=redact(result.stderr.decode()))
        else:
            entry.update(stdout_sha256=hashlib.sha256(result.stdout).hexdigest(), stderr=redact(result.stderr.decode()))
        commands.append(entry)
        return result.stdout
    except (OSError, subprocess.TimeoutExpired) as error:
        commands.append({'argv': args, 'cwd': str(cwd), 'exit_code': None, 'status': type(error).__name__, 'error': redact(str(error))})
        return b''


def save(name, value):
    path = OUT / name
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(value, indent=2, default=str) + '\n')


def source_bytes(name):
    return run(['git', 'show', SOURCE + ':' + name], retain=False)


def digest(data):
    return hashlib.sha256(data).hexdigest()
